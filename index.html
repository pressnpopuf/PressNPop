async function finalizeCartCheckout() {
    cart = cart.filter(
        item => item && item.name && Number(item.quantity) > 0
    );

    saveCart();

    if (!cart.length) {
        alert("Din varukorg är tom ♡");
        closeModal("cartCheckoutModal");
        return;
    }

    const firstName =
        document.getElementById("cartCheckoutFirstName").value.trim();

    const lastName =
        document.getElementById("cartCheckoutLastName").value.trim();

    const phone =
        document.getElementById("cartCheckoutPhone").value.trim();

    const email =
        document.getElementById("cartCheckoutEmail").value.trim();

    const delivery =
        document.getElementById("cartCheckoutDelivery").value;

    const address =
        document.getElementById("cartCheckoutAddress").value.trim();

    const postal =
        document.getElementById("cartCheckoutPostal").value.trim();

    const city =
        document.getElementById("cartCheckoutCity").value.trim();

    const message =
        document.getElementById("cartCheckoutMessage").value.trim();


    /* ---------------------------------------------
       Kontrollera kunduppgifter
    --------------------------------------------- */

    if (!firstName || !lastName || !phone || !email) {
        alert(
            "Fyll i förnamn, efternamn, telefonnummer och e-post ♡"
        );
        return;
    }


    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
        alert("Kontrollera e-postadressen ♡");
        return;
    }


    if (
        delivery === "shipping" &&
        (!address || !postal || !city)
    ) {
        alert(
            "Fyll i adress, postnummer och ort för frakt ♡"
        );
        return;
    }


    /* ---------------------------------------------
       Förbered ordern
    --------------------------------------------- */

    const orderCart = cart.map(item => ({
        ...item,
        options: {
            ...(item.options || {})
        }
    }));


    /* ---------------------------------------------
       Gör FormData
    --------------------------------------------- */

    const formData = new FormData();


    const payload = {
        firstName,
        lastName,
        phone,
        email,

        delivery:
            delivery === "shipping"
                ? "Frakt"
                : "Hämta upp",

        address:
            delivery === "shipping"
                ? `${address}, ${postal} ${city}`
                : "",

        message,

        cart: orderCart
    };


    formData.append(
        "payload",
        JSON.stringify(payload)
    );


    /* ---------------------------------------------
       Lägg med inspo-bilder
    --------------------------------------------- */

    for (
        let index = 0;
        index < orderCart.length;
        index++
    ) {

        const item = orderCart[index];

        if (
            item.options &&
            item.options.inspoData
        ) {

            try {

                const response =
                    await fetch(
                        item.options.inspoData
                    );

                const blob =
                    await response.blob();

                const extension =
                    blob.type === "image/png"
                        ? "png"
                        : blob.type === "image/webp"
                            ? "webp"
                            : "jpg";

                const fieldName =
                    `inspo_${index}`;

                formData.append(
                    fieldName,
                    blob,
                    `inspo-${index + 1}.${extension}`
                );

                item.options.inspoField =
                    fieldName;

            } catch (error) {

                console.warn(
                    "Kunde inte lägga till inspo-bild:",
                    error
                );
            }
        }
    }


    /* ---------------------------------------------
       Skicka automatiskt till Render
    --------------------------------------------- */

    const button =
        document.querySelector(
            "#cartCheckoutModal .step-buttons .knapp:not(.knapp-outline)"
        );


    if (button) {
        button.disabled = true;
        button.textContent =
            "Skickar beställning...";
    }


    try {

        const response =
            await fetch(
                "https://pnp-backend.onrender.com/api/orders",
                {
                    method: "POST",
                    body: formData
                }
            );


        let result = {};

        try {
            result = await response.json();
        } catch (error) {
            result = {};
        }


        if (!response.ok) {

            throw new Error(
                result.error ||
                "Kunde inte skicka beställningen."
            );
        }


        /* -----------------------------------------
           Lyckad beställning
        ----------------------------------------- */

        closeModal("cartCheckoutModal");

        cart = [];

        saveCart();

        updateCartUI();


        alert(
            `Tack för din beställning ♡\n\n` +
            `Ordernummer: ${result.orderNumber || "—"}\n` +
            `Totalt: ${Number(result.total || 0)
                .toFixed(2)
                .replace(".", ",")} kr\n\n` +
            `Beställningen har skickats till PressN'Pop.\n` +
            `Betala därefter via Swish.`
        );


    } catch (error) {

        console.error(
            "Beställningsfel:",
            error
        );


        alert(
            "Beställningen kunde inte skickas just nu.\n\n" +
            "Försök igen om en liten stund."
        );


    } finally {

        if (button) {

            button.disabled = false;

            button.textContent =
                "Lägg beställning →";
        }
    }
}

    function formatCartItemForEmail(item, index) {
        const lines = [];
        lines.push(`${index}. ${escapeEmailText(item.name)}`);
        lines.push(`   Antal: ${Number(item.quantity) || 1}`);
        if (item.color) lines.push(`   Färg: ${escapeEmailText(item.color)}`);
        if (item.length) lines.push(`   Längd: ${escapeEmailText(item.length)}`);

        const sizes = item.sizes || {};
        const labels = { thumb:"Tumme/Stortå", index:"Pekfinger/Tå 2", middle:"Långfinger/Tå 3", ring:"Ringfinger/Tå 4", pinky:"Lillfinger/Lilltå", toe:"Tå" };
        const sizeLines = Object.entries(sizes)
            .filter(([, value]) => value !== undefined && value !== null && String(value) !== "")
            .map(([key, value]) => `      ${labels[key] || key}: ${escapeEmailText(value)}`);
        if (sizeLines.length) {
            lines.push("   Storlekar:");
            lines.push(...sizeLines);
        }

        const options = item.options || {};
        if (options.form) lines.push(`   Form: ${escapeEmailText(options.form)}`);
        if (options.changes) lines.push(`   Ändringar: ${escapeEmailText(options.changes)}`);
        if (options.inspoFile) lines.push(`   Inspo-bild: ${escapeEmailText(options.inspoFile)} (bifoga bilden i mailet om den inte följer med automatiskt)`);
        return lines.join("\n");
    }

        if (delivery === "shipping" && (!address || !postal || !city)) {

    function getCartItemUnitPrice(item) {
        if (item.type === "custom_nails" || item.type === "custom_toes") return 299;

    }

        return cart.reduce((sum, item) => {

    function updateCartUI() {
        const count = cart.reduce((total, item) => total + Number(item.quantity || 0), 0);
        const countEl = document.getElementById("cartCount");
        const totalQuantityEl = document.getElementById("cartTotalQuantity");

        if (countEl) countEl.textContent = count;
        if (totalQuantityEl) totalQuantityEl.textContent = count;

        const container = document.getElementById("cartItems");
        const empty = document.getElementById("cartEmpty");
        const footer = document.getElementById("cartFooter");

        if (!container || !empty || !footer) return;

        if (!cart.length) {
            container.innerHTML = "";
            empty.style.display = "block";
            footer.style.display = "none";
            return;
        }

        empty.style.display = "none";
        footer.style.display = "block";

        container.innerHTML = cart.map(item => {
            const unitPrice = getCartItemUnitPrice(item);
            const itemTotal = unitPrice * Math.max(1, Number(item.quantity) || 1);
            const optionText = [
                item.color ? `Färg: ${item.color}` : "",
                item.length ? `Längd: ${item.length}` : "",
                item.options?.form ? `Form: ${item.options.form}` : ""
            ].filter(Boolean).join("<br>");

            return `
            <div class="cart-item">
                <img class="cart-item-image" src="${item.image}" alt="${item.name}">
                <div class="cart-item-info">
                    <h3>${item.name}</h3>
                    <p>${optionText}</p>
                    <p>Storlek: ${item.sizes?.thumb || item.sizes?.toe || "-"}/${item.sizes?.index || "-"}/${item.sizes?.middle || "-"}/${item.sizes?.ring || "-"}/${item.sizes?.pinky || "-"}</p>
                    <div class="cart-item-price">
                        <div><span class="cart-price-label">Pris/st:</span> ${unitPrice.toLocaleString("sv-SE")} kr</div>
                        <div><span class="cart-price-label">Summa:</span> <strong>${itemTotal.toLocaleString("sv-SE")} kr</strong></div>
                    </div>
                </div>
                <div class="cart-item-actions">
                    <button type="button" onclick="changeCartQuantity('${item.cartId}', -1)">−</button>
                    <strong>${item.quantity}</strong>
                    <button type="button" onclick="changeCartQuantity('${item.cartId}', 1)">+</button>
                    <button type="button" class="cart-remove" onclick="removeCartItem('${item.cartId}')">Ta bort</button>
                </div>
            </div>`;
        }).join("");

        updateCartTotals();
    }

        }
